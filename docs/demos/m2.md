To read the fixture deal's results:

```bash
cd packages/engine
cat > m2-demo.ts <<'TS'
import { run, explain } from './src/index';
import { fixtureDeal } from './test/fixtures';
const out = run(fixtureDeal());
const usd = (x: number | null) => (x === null ? 'n/a' : Math.round(x).toLocaleString('en-US'));
const pct = (x: number | null) => (x === null ? 'n/a' : (x * 100).toFixed(2) + '%');
console.log('TPC', usd(out.totals.tpc), 'per unit', usd(out.metrics.costPerUnit), 'equity', usd(out.totals.equity));
console.log('refinance', out.refinance && { loan: usd(out.refinance.loan), net: usd(out.refinance.net) });
console.log('sale', { value: usd(out.sale.value), net: usd(out.sale.netToEquity) });
console.log('yield on cost', pct(out.metrics.yieldOnCost), 'spread', pct(out.metrics.developmentSpread));
console.log('IRR project', pct(out.metrics.projectIrr), 'LP', pct(out.metrics.lpIrr), 'GP', pct(out.metrics.gpIrr), 'unlevered', pct(out.metrics.unleveredIrr));
console.log('multiple', out.metrics.equityMultiple?.toFixed(2), 'peak equity', usd(out.metrics.peakEquity));
console.log(out.annual.map((a) => `Y${a.year}${a.partial ? '*' : ''} NOI ${usd(a.noi)} DS ${usd(a.debtService.total)} DSCR ${a.dscr?.toFixed(2) ?? '-'}`).join('\n'));
console.log(explain(out, 'metrics.yieldOnCost'));
TS
../../apps/cli/node_modules/.bin/tsx m2-demo.ts; rm m2-demo.ts
```
